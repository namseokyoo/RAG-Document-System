```python
def evaluate_table_structure(ground_truth_table, vision_text):
    """
    표 구조 (행/열 개수, 헤더) 일치 확인
    """
    gt_rows = len(ground_truth_table.rows)
    gt_cols = len(ground_truth_table.columns)

    # Vision 분석에서 "N행 M열" 추출
    match = re.search(r'(\d+)행.*?(\d+)열', vision_text)
    if match:
        v_rows, v_cols = int(match.group(1)), int(match.group(2))
        return (gt_rows == v_rows) and (gt_cols == v_cols)
    return False
```
